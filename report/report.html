3. Fixtures

Використано autouse (_clean_environment прибирає змінні STUDENT_APP_*), scope=session (незмінний набір дозволених груп), fixture factory (student_factory з лічильником ID), autospec Mock (notifier), ланцюжок fixtures (service → populated_service) і fixture із yield output_dir, яка після тесту перевіряє, що атомарний export не лишив тимчасових файлів.

@pytest.fixture(autouse=True)
def _clean_environment(monkeypatch: pytest.MonkeyPatch) -> None:
    for name in ENV_NAMES:
        monkeypatch.delenv(ENV_PREFIX + name, raising=False)

@pytest.fixture(scope="session")
def allowed_groups() -> frozenset[str]:
    return frozenset({"PI-21", "PI-22", "PI-23"})

@pytest.fixture
def student_factory() -> StudentFactory:
    counter = itertools.count(1)

    def create(
        last_name: str = "Коваленко",
        first_name: str = "Олена",
        group: str = "PI-21",
        grades: tuple[float, ...] = (90.0,),
        student_id: int | None = None,
    ) -> Student:
        identifier = next(counter) if student_id is None else student_id
        return Student(identifier, first_name, last_name, group, grades)

    return create
@pytest.fixture
def notifier() -> Notifier:
    return create_autospec(Notifier, instance=True)

@pytest.fixture
def output_dir(tmp_path: Path) -> Iterator[Path]:
    directory = tmp_path / "output"
    yield directory
    leftovers = list(directory.glob("*.tmp")) if directory.exists() else []
    assert leftovers == [], f"тимчасові файли не видалено: {leftovers}"

def test_average_without_grades_raises_custom_error(student_factory: StudentFactory) -> None:

def test_average_tolerates_floating_point_noise(student_factory: StudentFactory) -> None:

def test_notifier_failure_does_not_roll_back_registration(
    service: StudentService,
    notifier: Mock,
    student_factory: StudentFactory,
    caplog: pytest.LogCaptureFixture,
) -> None:
    notifier.send.side_effect = NotificationError("SMTP недоступний")
    student = student_factory(student_id=6)
    with caplog.at_level(logging.WARNING):
        service.register(student)
    assert service.get(6) is student
    assert service.failed_notifications == [6]
    assert "Не вдалося сповістити студента 6" in caplog.text

def test_repository_failure_prevents_notification(
    student_factory: StudentFactory,
) -> None:
    repository = create_autospec(StudentRepository, instance=True)
    notifier = create_autospec(Notifier, instance=True)
    repository.add.side_effect = RepositoryError("сховище недоступне")
    service = StudentService(repository, notifier)
    with pytest.raises(RepositoryError):
        service.register(student_factory())
    notifier.send.assert_not_called()
    assert service.failed_notifications == []

def test_failed_write_keeps_previous_file_and_leaves_no_temporary_file(

5. AsyncMock

Асинхронний репозиторій – зовнішня залежність, тому AsyncStudentService тестується з create_autospec(AsyncStudentRepository), який створює AsyncMock для кожного async def. Перевіряються результат, assert_awaited_once_with, assert_not_awaited і збій мережі (side_effect=ConnectionError), що перетворюється на RepositoryError зі збереженою причиною. Режим pytest-asyncio – strict.

@pytest.fixture
def async_repository() -> AsyncMock:
    return create_autospec(AsyncStudentRepository, instance=True)

async def test_load_returns_student_from_async_repository(
    async_service: AsyncStudentService, async_repository: AsyncMock, student_factory: StudentFactory
) -> None:
    student = student_factory(student_id=1)
    async_repository.get.return_value = student
    assert await async_service.load(1) is student
    async_repository.get.assert_awaited_once_with(1)

async def test_connection_failure_becomes_repository_error_with_cause(
    async_service: AsyncStudentService, async_repository: AsyncMock
) -> None:
    async_repository.get.side_effect = ConnectionError("таймаут")
    with pytest.raises(RepositoryError, match="сховище недоступне") as error:
        await async_service.load(1)
    assert isinstance(error.value.__cause__, ConnectionError)

Mock доречний на системних межах (сповіщення, мережа), fake – для сховищ. Flaky-тест небезпечний тим, що підриває довіру до червоних збірок; випадковість треба фіксувати (seed, ін'єкція даних).

Висновки

Створено test suite із 195 тестів для системи обліку студентів (варіант 1): unit tests моделі, оцінювання, репозиторіїв, сервісів, звітів, конфігурації, CSV import і JSON export, async-сервісу та integration tests повного потоку CSV → валідація → звіт, CLI і async-потоку. Використано fixtures різних видів, параметризацію, pytest.raises, approx, tmp_path, monkeypatch, Mock/autospec, AsyncMock і fake. Негативні тести знайшли п'ять дефектів production code, які не бачив 100 % coverage; statement і branch coverage – 100.00 % при порозі 90 % у pyproject.toml. Експерименти показали, що coverage не замінює змістовних assertions, а детермінованість та ізоляція тестів важливіші за їх кількість.
